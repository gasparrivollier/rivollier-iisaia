# Slice 0.1 Scaffold Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stand up the Phase 0 web-MVP skeleton. That means a FastAPI backend with `/health` and a `users` migration, a fake Pl@ntNet server, a `compose.yaml` stack (Postgres + MinIO + backend + fake-plantnet) on podman, a Flutter Web build with semantics for testing, and a Playwright smoke test proving that the stack is healthy and the app shows the Diary.

**Architecture:** The backend is sync SQLAlchemy 2 + psycopg 3 + Alembic, managed with `uv` on Python 3.13. fake-plantnet is a tiny FastAPI app that replays `docs/examples/plantnet-identify-response.json`. Everything runs under `podman compose`. Playwright's globalSetup brings the stack up and polls the health endpoints. Its `webServer` statically serves `client/build/web`, built with `--dart-define=E2E=true` so Flutter emits a DOM semantics tree that Playwright can locate.

**Tech Stack:** Python 3.13, uv, FastAPI, SQLAlchemy 2, psycopg 3, Alembic, pydantic-settings, pytest, ruff; Postgres 17; MinIO (Chainguard image); Flutter 3.47 (web); Node 22 + Playwright (TypeScript); podman 5.8 + podman-compose 1.6.

**Spec:** `docs/superpowers/specs/2026-10-14-web-mvp-design.md` (slice 0.1 row). Also read `docs/architecture.md` ("Repo layout", "Data model", "Dev and test tooling").

## Review Focus

1. **A real key in `.env` must not leak into tests.** Compose interpolates the root `.env`. If that file ever sets `PLANTNET_BASE_URL` to the real API, e2e runs would spend quota. The guard has two parts: the compose default points at fake-plantnet (tested in Task 3 by `test_compose_config.py`), and e2e globalSetup forces `PLANTNET_BASE_URL=http://fake-plantnet:8100` in the environment it passes to compose (Task 5).
2. **DB unreachable → `/health` returns 503 fast**, not a 500 or a hang. A connect timeout of 3s is set on the engine (tested in Task 2 by `test_health_returns_503_quickly_when_db_unreachable`).
3. **Backend container starts before Postgres is ready** → it retries instead of crash-looping (`app.wait_for_db`, tested in Task 2).
4. **Re-running `compose up` on existing volumes** → migrations are idempotent (`test_upgrade_head_twice_is_noop`, Task 2).
5. **The `E2E` dart-define must not change normal builds or widget tests.** With the flag off, `flutter test` (existing `app_flow_test.dart`) still passes, and the semantics identifier is testable in a widget test without the flag (Task 4).

```python
from collections.abc import Iterator

from sqlalchemy import Engine, create_engine
from sqlalchemy.orm import DeclarativeBase, Session, sessionmaker

from app.config import get_settings


class Base(DeclarativeBase):
    pass


def make_engine(url: str) -> Engine:
    # connect_timeout keeps /health from hanging when Postgres is down.
    return create_engine(url, pool_pre_ping=True, connect_args={"connect_timeout": 3})


engine = make_engine(get_settings().database_url)  # lazy: does not connect until first use
SessionLocal = sessionmaker(bind=engine, autoflush=False, expire_on_commit=False)


def get_db() -> Iterator[Session]:
    with SessionLocal() as session:
        yield session
```

`backend/app/wait_for_db.py`:

```python
"""Block until Postgres accepts connections. Run as `python -m app.wait_for_db` before migrations."""

import time

from sqlalchemy import text
from sqlalchemy.exc import OperationalError

from app.config import get_settings
from app.db import make_engine


def wait_for_db(url: str, attempts: int = 30, delay: float = 1.0) -> None:
    engine = make_engine(url)
    try:
        for attempt in range(1, attempts + 1):
            try:
                with engine.connect() as conn:
                    conn.execute(text("SELECT 1"))
                return
            except OperationalError:
                if attempt == attempts:
                    raise
                print(f"wait_for_db: Postgres not ready (attempt {attempt}/{attempts})", flush=True)
                time.sleep(delay)
    finally:
        engine.dispose()
```

`fake-plantnet/server.py`:

```python
"""Fake Pl@ntNet API for tests: replays a recorded /v2/identify response.

Never spends real quota. Point the backend's PLANTNET_BASE_URL here.
"""

import copy
import json
import os
from datetime import UTC, datetime, timedelta
from email.utils import format_datetime
from pathlib import Path

from fastapi import FastAPI, Query, Request
from fastapi.responses import JSONResponse
from pydantic import BaseModel

DEFAULT_RECORDED = (
    Path(__file__).resolve().parents[1] / "docs/examples/plantnet-identify-response.json"
)


class NextStatus(BaseModel):
    status: int


def _error(status: int, message: str, headers: dict[str, str] | None = None) -> JSONResponse:
    return JSONResponse(
        status_code=status,
        content={"statusCode": status, "error": "Fake", "message": message},
        headers=headers,
    )


def _next_utc_midnight() -> str:
    now = datetime.now(UTC)
    midnight = (now + timedelta(days=1)).replace(hour=0, minute=0, second=0, microsecond=0)
    return format_datetime(midnight, usegmt=True)


def create_app(recorded_path: Path | None = None) -> FastAPI:
    path = recorded_path or Path(os.environ.get("RECORDED_RESPONSE", DEFAULT_RECORDED))
    recorded = json.loads(path.read_text())
    state: dict[str, int | None] = {"next_status": None}
    app = FastAPI(title="fake-plantnet")

    @app.get("/health")
    def health() -> dict[str, str]:
        return {"status": "ok"}

    @app.post("/_control/next-status")
    def set_next_status(body: NextStatus) -> dict[str, int]:
        state["next_status"] = body.status
        return {"next_status": body.status}

    @app.post("/v2/identify/{project}")
    async def identify(
        project: str, request: Request, api_key: str | None = Query(None, alias="api-key")
    ):
        if not api_key:
            return _error(401, "Missing api-key")
        forced, state["next_status"] = state["next_status"], None
        if forced == 429:
            return _error(429, "Too Many Requests", {"Retry-After": _next_utc_midnight()})
        if forced is not None:
            return _error(forced, "Forced by test")
        form = await request.form()
        if not form.getlist("images"):
            return _error(400, "At least one image is required")
        body = copy.deepcopy(recorded)
        body["query"]["project"] = project
        return JSONResponse(
            body,
            headers={
                "RateLimit-Policy": '"identify";q=500;w=86400',
                "RateLimit": '"identify";r=499;t=3600',
            },
        )

    return app


app = create_app()
```

```ts
import path from 'node:path';

import { BACKEND_URL, FAKE_PLANTNET_URL, MINIO_URL } from './support/urls';

const ROOT = path.resolve(__dirname, '..');

async function waitFor(url: string, timeoutMs: number): Promise<void> {
  const deadline = Date.now() + timeoutMs;
  let last = '';
  while (Date.now() < deadline) {
    try {
      const res = await fetch(url);
      if (res.ok) return;
      last = `HTTP ${res.status}`;
    } catch (err) {
      last = String(err);
    }
    await new Promise((r) => setTimeout(r, 1000));
  }
  throw new Error(`Timed out waiting for ${url} (last: ${last})`);
}

export default async function globalSetup(): Promise<void> {
  if (!process.env.E2E_SKIP_COMPOSE) {
    execSync('podman compose up -d --build', {
      cwd: ROOT,

      env: { ...process.env, PLANTNET_BASE_URL: 'http://fake-plantnet:8100' },
    });

  await waitFor(`${FAKE_PLANTNET_URL}/health`, 60_000);
  await waitFor(`${MINIO_URL}/minio/health/live`, 60_000);
}
```
